	possible = {
		custom_tooltip = {
			text = "poland_viceroy_cooldown_tt"
			NOT = { has_global_variable = poland_viceroy_cooldown }
		}
		trigger_if = {
			limit = {
				scope:congress_poland_scope ?= THIS
			}
			custom_tooltip = {
				text = poland_already_conciliatory
				ruler = {
					NOT = { has_variable = poland_conciliatory }
				}
			}
			custom_tooltip = {
				text = poland_viceroy_bar_trigger
				je:je_congress_poland = {
					scripted_bar_progress ={
						name = je_congress_poland_progress_bar
						value <= congress_poland_viceroy_request_threshold
					}
				}
			}
		}
	}

	ai_chance = {
		value = 0
		# Poland does this if the tsar might be amenable, unless it backs the Reds
		if = {
			limit = {
				scope:congress_poland_scope ?= this
				NOT = { has_variable = ai_poland_red }
				scope:russia_poland_scope = {
					ruler = {
						tsar_reactionary_ideology_trigger = no
					}
				}
			}
			add = 100
		}
		# Russia appoints a nicer governor once the Reds grow too strong
		if = {
			limit = {
				scope:russia_poland_scope ?= THIS
				ai_russia_should_appease_poland = yes
				ai_russia_poland_bar_trending_white = no
				scope:congress_poland_scope = {
					ruler = {
						NOT = { has_variable = poland_conciliatory }
					}
				}
			}
			add = 100
		}
		# An intransigent Russia brings back a harsher governor once the Reds are contained
		if = {
			limit = {
				scope:russia_poland_scope ?= THIS
				ai_russia_should_appease_poland = no
				ai_russia_is_intransigent_on_poland = yes
				congress_poland_bar_value <= 0
				scope:congress_poland_scope = {
					ruler = {
						has_variable = poland_conciliatory
					}
				}
			}
			add = 100
		}
	}

	effect = {
		set_global_variable = {
			name = poland_viceroy_cooldown
			days = normal_modifier_time
		}
		scope:russia_poland_scope = {
			trigger_event = {
				id = congress_poland.1
				days = 0
				popup = yes
			}
		}
	}
}
